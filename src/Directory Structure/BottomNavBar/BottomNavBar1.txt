import React, { useEffect, useState } from 'react';
import { View, TouchableOpacity, StyleSheet, Platform, } from 'react-native';
import MaterialCommunityIcons from 'react-native-vector-icons/MaterialCommunityIcons';
import { useNavigation, useRoute } from '@react-navigation/native';
import { NativeStackNavigationProp, NativeStackScreenProps } from '@react-navigation/native-stack';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { useSafeAreaInsets } from 'react-native-safe-area-context';
import { RootStackParamList } from './AppNavigatorProduct';
import { User } from './UserDatabase';

type NavigationPropType = NativeStackNavigationProp<RootStackParamList>;
type RoutePropType = NativeStackScreenProps<RootStackParamList>['route'];

const BottomNavBar = () => {
  const navigation = useNavigation<NavigationPropType>();
  const route = useRoute<RoutePropType>();
  const insets = useSafeAreaInsets();
  const [user, setUser] = useState<User | null>(null);

  useEffect(() => {
    const loadUser = async () => {
      try {
        const userData = await AsyncStorage.getItem('user');
        if (userData) {
          setUser(JSON.parse(userData));
        } else {
          setUser(null);
        }
      } catch (error) {
        console.error('Error loading user from AsyncStorage:', error);
      }
    };
    loadUser();
  }, []);

  const isActive = (screen: keyof RootStackParamList) => route.name === screen;

  // Xác định chế độ điều hướng: gesture navigation khi insets.bottom === 0
  const isGestureNavigation = insets.bottom === 0;

  return (
    <View
      style={[
        styles.wrapper,
        {
          height: isGestureNavigation ? 40 + 10 : 60 + insets.bottom, // Giảm chiều cao wrapper
        },
      ]}
    >
      <View
        style={[
          styles.container,
          {
            bottom: isGestureNavigation ? 10 : insets.bottom, // Vị trí navbar
            height: isGestureNavigation ? 40 : 60, // Giảm chiều cao navbar
          },
        ]}
      >
        <TouchableOpacity
          style={[styles.tab, { height: isGestureNavigation ? 40 : 60 }]}
          onPress={() => navigation.navigate({ name: 'HomeScreen', params: undefined })}
        >
          <View
            style={[
              styles.tabBackground,
              { height: isGestureNavigation ? 40 + 10 : 60 + insets.bottom }, // Màu nền tràn xuống
              isActive('HomeScreen') && styles.activeTab
            ]}
          />
          <MaterialCommunityIcons
            name="home"
            size={26} // Giảm kích thước icon cho navbar nhỏ hơn
            color={isActive('HomeScreen') ? '#4DB6AC' : '#888'}
            style={styles.icon}
          />
        </TouchableOpacity>
        <TouchableOpacity
          style={[styles.tab, { height: isGestureNavigation ? 40 : 60 }]}
          onPress={() => navigation.navigate({ name: 'CategoryScreen', params: undefined })}
        >
          <View
            style={[
              styles.tabBackground,
              { height: isGestureNavigation ? 40 + 10 : 60 + insets.bottom }, // Màu nền tràn xuống
              isActive('CategoryScreen') && styles.activeTab
            ]}
          />
          <MaterialCommunityIcons
            name="shopping"
            size={26} // Giảm kích thước icon
            color={isActive('CategoryScreen') ? '#4DB6AC' : '#888'}
            style={styles.icon}
          />
        </TouchableOpacity>
        <TouchableOpacity
          style={[styles.tab, { height: isGestureNavigation ? 40 : 60 }]}
          onPress={() => navigation.navigate({ name: 'ProfileScreen', params: {} })}
        >
          <View
            style={[
              styles.tabBackground,
              { height: isGestureNavigation ? 40 + 10 : 60 + insets.bottom }, // Màu nền tràn xuống
              isActive('ProfileScreen') && styles.activeTab
            ]}
          />
          <MaterialCommunityIcons
            name="account"
            size={26} // Giảm kích thước icon
            color={isActive('ProfileScreen') ? '#4DB6AC' : '#888'}
            style={styles.icon}
          />
        </TouchableOpacity>
      </View>
    </View>
  );
};

const styles = StyleSheet.create({
  wrapper: {
    position: 'absolute',
    bottom: 0,
    left: 0,
    right: 0,
    backgroundColor: '#fff', // Màu nền trắng làm nền mặc định
  },
  container: {
    flexDirection: 'row',
    borderTopWidth: 1,
    borderTopColor: '#eee',
    elevation: 5,
    shadowColor: '#000',
    shadowOpacity: 0.1,
    shadowOffset: { width: 0, height: -2 },
    shadowRadius: 4,
    position: 'absolute',
    left: 0,
    right: 0,
  },
  tab: {
    flex: 1,
    justifyContent: 'center',
    alignItems: 'center',
  },
  tabBackground: {
    position: 'absolute',
    top: 0,
    left: 0,
    right: 0,
    backgroundColor: '#fff', // Màu nền trắng cho tab không active
  },
  activeTab: {
    backgroundColor: '#e0f7f5', // Màu nền cho tab active
  },
  icon: {
    // Không cần marginTop, icon được căn giữa trong container
  },
});

export default BottomNavBar;
